/**
 * Función principal para AppSheet
 */
function doGet(e) {
  try {
    const buscarParametro = (nombre) => {
      if (e.parameter[nombre]) return e.parameter[nombre];
      for (let key in e.parameter) {
        if (key.toLowerCase().indexOf(nombre.toLowerCase()) !== -1) {
          return e.parameter[key];
        }
      }
      return "";
    };

    // Recolectamos todos los parámetros enviados desde la URL de AppSheet
    const data = {
      Fecha: buscarParametro("Fecha"),
      FacturaProveedor: buscarParametro("FacturaProveedor"),
      ProveedorNombre: buscarParametro("ProveedorNombre"),
      DiasCredito: buscarParametro("DiasCredito"),
      Vencimiento: buscarParametro("Vencimiento"),
      Direccion: buscarParametro("Direccion"),
      Colonia: buscarParametro("Colonia"),
      Ciudad: buscarParametro("Ciudad"),
      Moneda: buscarParametro("Moneda"),
      FilasDatos: buscarParametro("FilasDatos"), // Cadena comprimida separada por ^ y |
      TotalCobrado: buscarParametro("TotalCobrado") || "0",
      TotalProveedor: buscarParametro("TotalProveedor") || "0",
      TotalUtilidad: buscarParametro("TotalUtilidad") || "0"
    };

    const fileData = generatePDF(data);
    
    const htmlOutput = `
      <!DOCTYPE html>
      <html>
      <head>
        <title>Descarga Factura Roelca</title>
        <style>
            body { font-family: Arial, sans-serif; text-align: center; padding-top: 50px; background-color: #f9f9f9; }
            h2 { color: #00AEEF; }
            p { color: #666; font-size: 14px; }
            .btn-descarga { 
                display: inline-block; margin-top: 20px; padding: 12px 24px; 
                background-color: #F15A24; color: #fff; text-decoration: none; 
                font-weight: bold; border-radius: 5px; 
            }
        </style>
      </head>
      <body>
        <h2>Generando Documento...</h2>
        <p>Tu documento debería descargarse automáticamente.</p>
        <p>Si la descarga no inicia, haz clic en el botón:</p>
        
        <a id="btn-descarga" class="btn-descarga" href="data:application/pdf;base64,${fileData.base64}" download="${fileData.fileName}">
          Descargar PDF Manualmente
        </a>

        <script>
          setTimeout(function() {
            document.getElementById('btn-descarga').click();
          }, 800);
        </script>
      </body>
      </html>
    `;

    return HtmlService.createHtmlOutput(htmlOutput);
    
  } catch (error) {
    return ContentService.createTextOutput("Error en el script: " + error.toString());
  }
}

/**
 * Función que construye el HTML y genera el PDF

function generatePDF(data) {
  const folderId = "1gPGw8KCyQqY6Uj8U6fKq9S-dsSvtFieF"; 
  const folder = DriveApp.getFolderById(folderId);
  
  // --- FUNCIÓN PARA FORMATEAR MONEDA ---
  const formatearMoneda = (val) => {
    if (!val || val.toString().trim() === "") return "0.00";
    let str = val.toString().trim();
    
    // Si tiene punto decimal, evaluamos cuántos decimales tiene
    if (str.includes('.')) {
      let partes = str.split('.');
      if (partes[1].length === 1) {
        return str + '0'; // Ejemplo: .4 pasa a ser .40
      }
      return str; // Si tiene 2 o más decimales, se queda igual
    } else {
      return str + '.00'; // Si es un número entero sin punto, le agregamos .00

  };

  // Aplicar formato a los totales generales
  const totalCobradoFormat = formatearMoneda(data.TotalCobrado);
  const totalProveedorFormat = formatearMoneda(data.TotalProveedor);
  const totalUtilidadFormat = formatearMoneda(data.TotalUtilidad);

  // --- PROCESAMIENTO DE LA IMAGEN LOGO A BASE64 ---
  const imageId = "1blNDWMQvvp7Xz3G7lm_whAxQw9krUOAz";
  let base64Image = "";
  try {
    const imgBlob = DriveApp.getFileById(imageId).getBlob();
    base64Image = "data:" + imgBlob.getContentType() + ";base64," + Utilities.base64Encode(imgBlob.getBytes());
  } catch (e) { 
    console.log("Error con la imagen: " + e);

    const filas = data.FilasDatos.split("^"); 
    filas.forEach(fila => {
      const col = fila.split("|"); 
      // Se asume el orden: REF | EQ | ORIGEN | DESTINO | ID_TARIFARIO | FACTURA | COBRADO | PROVEEDOR | UTILIDAD
      if (col.length >= 9) {
        // Formatear columnas de monedas antes de inyectarlas

        const proveedorRow = formatearMoneda(col[7]);
        const utilidadRow = formatearMoneda(col[8]);
