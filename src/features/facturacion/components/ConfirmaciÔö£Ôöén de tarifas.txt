/**
 * Función principal para AppSheet
 */
function doGet(e) {
  try {
    const buscarParametro = (nombre) => {
      if (e.parameter[nombre]) return e.parameter[nombre];
      for (let key in e.parameter) {
        if (key.toLowerCase().indexOf(nombre.toLowerCase()) !== -1) {
          return e.parameter[key];
        }
      }
      return "";
    };

    const data = {
      Coordinador: buscarParametro("Coordinador"),
      Referencia: buscarParametro("Referencia") || "S/R",
      Remolque: buscarParametro("Remolque"),
      TipoUnidad: buscarParametro("TipoUnidad"),
      PlacasRemolque: buscarParametro("PlacasRemolque"),
      Unidad: buscarParametro("Unidad"),
      Operador: buscarParametro("Operador"),
      FechaServicio: buscarParametro("FechaServicio"),
      TipoCambio: buscarParametro("TipoCambio"),
      Proveedor: buscarParametro("Proveedor"),
      TipoOperacion: buscarParametro("TipoOperacion"),
      ImpoExpoMov: buscarParametro("ImpoExpoMov"),

      ClienteOrigen: buscarParametro("ClienteOrigen"),
      CiudadOrigen: buscarParametro("CiudadOrigen"),
      ClienteDestino: buscarParametro("ClienteDestino"),
      CiudadDestino: buscarParametro("CiudadDestino"),
      RefCliente: buscarParametro("RefCliente"),
      Moneda: buscarParametro("Moneda"),
      MonedaConvenio: buscarParametro("MonedaConvenio"),
      DescripcionDireccion: buscarParametro("DescripcionDireccion"),
      CityStateZip: buscarParametro("CityStateZip"),
      ConvenioProv: buscarParametro("ConvenioProv"),
      CostosAdic: buscarParametro("CostosAdic"),
      SubtotalProv: buscarParametro("SubtotalProv"),
      TotalAPagar: buscarParametro("TotalAPagar")
    };

    const pdfUrl = generatePDF(data);
    
    return HtmlService.createHtmlOutput(
      "<html><body style='font-family: Arial; text-align: center; padding-top: 50px;'>" +
      "<h2>Generando Confirmación de Tarifa...</h2>" +
      "<script>window.location.replace('" + pdfUrl + "');</script></body></html>"
    );
    
  } catch (error) {
    return ContentService.createTextOutput("Error en el script: " + error.toString());
  }
}
